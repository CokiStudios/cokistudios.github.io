oauth-flow.js
// 1. Genera el PKCE challenge
const verifier = generateCode(128);
const challenge = await sha256(verifier);

// 2. Construye la URL de autorización
const url = new URL(
  'https://cokistudios.github.io/
   coki-authorize.html'
);
url.searchParams.set('client_id', MY_CLIENT_ID);
url.searchParams.set('redirect_uri', CALLBACK_URL);
url.searchParams.set('code_challenge', challenge);

// 3. Redirige al usuario
window.location.href = url.toString();
